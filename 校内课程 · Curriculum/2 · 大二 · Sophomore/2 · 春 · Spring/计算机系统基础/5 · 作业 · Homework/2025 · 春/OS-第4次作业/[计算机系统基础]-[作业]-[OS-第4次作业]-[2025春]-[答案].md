## 1

### 题目

三个进程 P1、P2、P3 互斥使用一个包含 N（N>0）个单元的缓冲区。P1 每次用 produce() 生成一个正整数并用 put() 送入缓冲区某一个空单元中；P2 每次用 getodd() 从该缓冲区中取出一个奇数并用 countodd() 统计奇数个数；P3 每次用 geteven() 从该缓冲区中取出一个偶数并用 counteven() 统计偶数个数。请用信号量机制实现这三个进程的同步与互斥活动，并说明所定义的信号量的含义。

### 答案

**答案：**

三个进程共享缓冲区，对缓冲区需要添加信号量进行保护，为 mutex，且初值为 1；缓冲区中空位的个数为三个进程共享，设信号量为 empty，初始值为 N。P1 和 P2 共享缓冲区中奇数的个数，奇数的个数需要信号量，设为 odd，初值为 0；P1 和 P3 共享缓冲区中的偶数个数，记偶数的个数信号量为 even，初值为 1。

```c
Semaphore mutex = 1, odd = 0, even = 1, empty = N

P1:
while (true) {
    P(empty);
    integer = produce();
    P(mutex);
    put();
    if (integer % 2 == 0) {
        V(even);
    } else {
        V(odd);
    }
    V(mutex);
}

P2:
while (true) {
    P(odd);
    P(mutex);
    getodd();
    V(mutex);
    V(empty);
    countodd();
}

P3:
while (true) {
    P(mutex);
    P(even);
    geteven();
    V(mutex);
    V(empty);
    counteven();
}
```

### 题目

一个野人部落从一个大锅中一起吃炖肉，这个大锅一次可以存放 M 人份的炖肉。当野人们想吃的时候，如果锅中不空，他们就自助着从大锅中吃肉。如果大锅空了，他们就唤醒厨师，等待厨师再做一锅肉。

野人线程未同步的代码如下：

```c
while (true) {
    getServingFromPot();
    eat();
}
```

厨师线程未同步的代码如下：

```c
while (true) {
    putServingsInPot(M);
}
```

同步的要求是：当大锅空的时候，野人不能够调用 getServingFromPot()；仅当大锅为空的时候，大厨才能够调用 putServingsInPot()。

问题：请写出使用 PV 满足同步要求的完整程序。

### 答案

**答案：**

设置信号量 empty = 1 表示没有食物，野人等待；设置 Have = 1 表示有食物，厨师等待；设置肉的数量为 food = 0。

```c
Semaphore empty = 1, Have = 1, food = 0

savages:
while (true) {
    P(Have);
    getServingFromPot();
    food--;
    if (food == 0)
        V(empty);
    else {
        V(Have);
    }
    eat();
}

cook:
while (true) {
    P(empty);
    putServingsInPot(M);
    food += M;
    V(Have);
}
```

### 题目

系统中有多个生产者进程和消费者进程，共享用一个可以存 1000 个产品的缓冲区（初始为空），当缓冲区为未满时，生产者进程可以放入一件其生产的产品，否则等待；当缓冲区为未空时，消费者进程可以取走一件产品，否则等待。要求一个消费者进程从缓冲区连续取出 10 件产品后，其他消费者进程才可以取产品，请用信号量 P、V 操作实现进程间的互斥和同步，要求写出完整的过程；并指出所用信号量的含义和初值。

### 答案

**答案：**

mutex 表示是否占用缓冲区；empty 表示缓冲区空余位置的个数，初始 empty = 1000；ProductNum 表示缓冲区产品数量，初始为 0；NoConsumer 表示缓冲区中没有消费者，初始为 1。

```c
Semaphore mutex = 1, empty = 1000, ProductNum = 0, NoConsumer = 1

Producer:
while (true) {
    P(empty);
    P(mutex);
    produce();
    V(mutex);
    V(ProductNum);
}

Consumer:
while (true) {
    P(NoConsumer);
    for (int i = 0; i < 10; i++) {
        P(ProductNum);
        P(mutex);
        consume();
        V(mutex);
        V(empty);
    }
    V(NoConsumer);
}
```

## 4

### 题目

读者写者问题（写者优先）：

1. 共享读；
2. 互斥写、读写互斥；
3. 写者优先于读者（一旦有写者，则后续读者必须等待，唤醒时优先考虑写者）。

### 答案

**答案：**

```c++
#include <pthread.h>
#include <stdio.h>

// 定义读写锁结构
typedef struct {
    pthread_mutex_t mutex;          // 保护共享状态的互斥锁
    pthread_cond_t read_cond;       // 读者等待的条件变量
    pthread_cond_t write_cond;      // 写者等待的条件变量
    int readers_active;             // 当前正在读的读者数
    int writers_waiting;            // 等待中的写者数
    int writing;                    // 写者是否正在写（1为是，0为否）
} rwlock_t;

// 初始化读写锁
void rwlock_init(rwlock_t *rw) {
    pthread_mutex_init(&rw->mutex, NULL);
    pthread_cond_init(&rw->read_cond, NULL);
    pthread_cond_init(&rw->write_cond, NULL);
    rw->readers_active = 0;
    rw->writers_waiting = 0;
    rw->writing = 0;
}

// 读者加锁
void reader_lock(rwlock_t *rw) {
    pthread_mutex_lock(&rw->mutex);
    // 如果有写者等待或正在写，读者等待
    while (rw->writers_waiting > 0 || rw->writing) {
        pthread_cond_wait(&rw->read_cond, &rw->mutex);
    }
    rw->readers_active++; // 增加活跃读者数
    pthread_mutex_unlock(&rw->mutex);
}

// 读者解锁
void reader_unlock(rwlock_t *rw) {
    pthread_mutex_lock(&rw->mutex);
    rw->readers_active--; // 减少活跃读者数
    // 如果没有读者了，尝试唤醒写者
    if (rw->readers_active == 0) {
        pthread_cond_signal(&rw->write_cond);
    }
    pthread_mutex_unlock(&rw->mutex);
}

// 写者加锁
void writer_lock(rwlock_t *rw) {
    pthread_mutex_lock(&rw->mutex);
    rw->writers_waiting++; // 增加等待的写者数
    // 如果有读者活跃或正在写，写者等待
    while (rw->readers_active > 0 || rw->writing) {
        pthread_cond_wait(&rw->write_cond, &rw->mutex);
    }
    rw->writers_waiting--; // 减少等待的写者数
    rw->writing = 1;       // 标记写者开始写
    pthread_mutex_unlock(&rw->mutex);
}

// 写者解锁
void writer_unlock(rwlock_t *rw) {
    pthread_mutex_lock(&rw->mutex);
    rw->writing = 0; // 标记写者结束
    // 优先唤醒等待的写者
    if (rw->writers_waiting > 0) {
        pthread_cond_signal(&rw->write_cond);
    }
    // 如果没有写者等待，再唤醒所有读者
    else {
        pthread_cond_broadcast(&rw->read_cond);
    }
    pthread_mutex_unlock(&rw->mutex);
}

//示例
rwlock_t rwlock;

void* reader_thread(void* arg) {
    reader_lock(&rwlock);
    // 读取操作
    printf("Reader %ld is reading\n", (long)arg);
    reader_unlock(&rwlock);
    return NULL;
}

void* writer_thread(void* arg) {
    writer_lock(&rwlock);
    // 写入操作
    printf("Writer %ld is writing\n", (long)arg);
    writer_unlock(&rwlock);
    return NULL;
}

int main() {
    pthread_t readers[5], writers[5];
    rwlock_init(&rwlock);

    // 创建2个写者和3个读者
    for (long i = 0; i < 2; i++) {
        pthread_create(&writers[i], NULL, writer_thread, (void*)i);
    }
    for (long i = 0; i < 3; i++) {
        pthread_create(&readers[i], NULL, reader_thread, (void*)i);
    }

    // 等待所有线程完成
    for (int i = 0; i < 2; i++) {
        pthread_join(writers[i], NULL);
    }
    for (int i = 0; i < 3; i++) {
        pthread_join(readers[i], NULL);
    }

    return 0;
}

```

### 题目

寿司店问题。假设一个寿司店有 5 个座位，如果你到达的时候有一个空座位，你可以立刻就坐。但是如果你到达的时候 5 个座位都是满的有人已经就坐，这就意味着这些人都是一起来吃饭的，那么你需要等待所有的人一起离开才能就坐。编写同步原语，实现这个场景的约束。

### 答案

**答案：**

```c++
#include <pthread.h>
#include <stdbool.h>

#define NUM_SEATS 5  // 寿司店座位数

// 寿司店状态管理结构
typedef struct {
    pthread_mutex_t lock;      // 共享状态保护锁
    pthread_cond_t all_left;   // 等待所有食客离开的条件变量
    pthread_cond_t can_seat;   // 等待安排新群体入座的条件变量
    int seats_occupied;        // 当前已占用座位数
    int waiting_customers;     // 正在等待的顾客数量
    bool is_waiting_group;     // 当前是否在安排等待群体入座
} SushiBar;

// 初始化寿司店状态
void sushi_bar_init(SushiBar *bar) {
    pthread_mutex_init(&bar->lock, NULL);
    pthread_cond_init(&bar->all_left, NULL);
    pthread_cond_init(&bar->can_seat, NULL);
    bar->seats_occupied = 0;
    bar->waiting_customers = 0;
    bar->is_waiting_group = false;
}

// 顾客进店逻辑
void take_seat(SushiBar *bar) {
    pthread_mutex_lock(&bar->lock);

    // 情况1: 直接入座
    if (bar->seats_occupied > 0 && bar->seats_occupied < NUM_SEATS) {
        bar->seats_occupied++;
        pthread_mutex_unlock(&bar->lock);
        return;
    }

    // 情况2: 需要等待
    bar->waiting_customers++;
    while (true) {
        if (bar->seats_occupied == 0) {
            // 步骤1: 开始安排新群体入座
            bar->is_waiting_group = true;
            bar->seats_occupied = 0;

            // 步骤2: 将当前等待顾客转为新群体
            int group_size = bar->waiting_customers < NUM_SEATS ?
                             bar->waiting_customers : NUM_SEATS;
            bar->waiting_customers -= group_size;
            bar->seats_occupied = group_size;
            bar->is_waiting_group = false;

            pthread_mutex_unlock(&bar->lock);
            return;
        }

        // 步骤3: 等待所有食客离开
        pthread_cond_wait(&bar->all_left, &bar->lock);
    }
}

// 顾客离开逻辑
void leave_seat(SushiBar *bar) {
    pthread_mutex_lock(&bar->lock);

    bar->seats_occupied--;

    // 所有食客已离开，且有人等待
    if (bar->seats_occupied == 0 && bar->waiting_customers > 0) {
        pthread_cond_broadcast(&bar->all_left);
    }

    pthread_mutex_unlock(&bar->lock);
}

//示例
SushiBar sushi_bar;

void* customer_thread(void* arg) {
    long cust_id = (long)arg;

    take_seat(&sushi_bar);
    printf("顾客%ld入座用餐\n", cust_id);

    // 模拟用餐时间
    sleep(rand() % 3);

    printf("顾客%ld离开\n", cust_id);
    leave_seat(&sushi_bar);

    return NULL;
}

int main() {
    sushi_bar_init(&sushi_bar);
    pthread_t threads[10];

    // 创建10个顾客线程
    for (long i = 1; i <= 10; i++) {
        sleep(rand() % 2); // 随机间隔到达
        pthread_create(&threads[i-1], NULL, customer_thread, (void*)i);
    }

    // 等待所有顾客结束
    for (int i = 0; i < 10; i++) {
        pthread_join(threads[i], NULL);
    }

    return 0;
}
```

### 题目

进门问题。

1. 请给出 P、V 操作和信号量的物理意义。
2. 一个软件公司有 5 名员工，每人刷卡上班。员工刷卡后需要等待，直到所有员工都刷卡后才能进入公司。为了避免拥挤，公司要求员工一个一个通过大门。所有员工都进入后，最后进入的员工负责关门。请用 P、V 操作实现员工之间的同步关系。

### 答案

**答案：**

   （1）请给出P、V操作和信号量的物理意义。

   信号量物理意义：信号量是一个整型计数器，表示可用资源的数量

   P 操作物理意义：申请资源/进入临界区

   V 操作物理意义：释放资源/退出临界区

   （2）一个软件公司有5名员工，每人刷卡上班。员工刷卡后需要等待，直到所有员工都刷卡后才能进入公司。为了避免拥挤，公司要求员工一个一个通过大门。所有员工都进入后，最后进入的员工负责关门。请用P、V操作实现员工之间的同步关系。

```c++
#include <semaphore.h>
#include <pthread.h>
#include <stdio.h>

#define NUM_EMPLOYEES 5

// 全局同步变量
sem_t mutex;           // 互斥锁（用于保护共享资源）
sem_t all_here;        // 全员到齐信号
sem_t entry_mutex;     // 进门互斥信号
int swipe_count = 0;    // 刷卡计数
int entry_count = 0;    // 进门计数

void* employee(void* id_ptr) {
    int id = (int)(long)id_ptr;

    // 1. 刷卡操作
    sem_wait(&mutex);
    swipe_count++;
    printf("员工 %d 刷卡\n", id);

    // 检查是否全员到齐
    if (swipe_count == NUM_EMPLOYEES) {
        // 全员已刷卡，唤醒其他员工
        for (int i = 0; i < NUM_EMPLOYEES - 1; i++) {
            sem_post(&all_here);
        }
    }
    sem_post(&mutex);

    // 2. 等待全员到齐（除最后一个员工外）
    if (id != NUM_EMPLOYEES) {
        sem_wait(&all_here);
    }

    // 3. 依次进门
    sem_wait(&entry_mutex);
    entry_count++;
    printf("员工 %d 进入公司\n", id);

    // 4. 最后进入的员工负责关门
    if (entry_count == NUM_EMPLOYEES) {
        printf("--> 员工 %d 关闭大门\n", id);

        // 重置计数器（为第二天准备）
        swipe_count = 0;
        entry_count = 0;
    }
    sem_post(&entry_mutex);

    return NULL;
}

int main() {
    // 初始化信号量
    sem_init(&mutex, 0, 1);
    sem_init(&all_here, 0, 0);
    sem_init(&entry_mutex, 0, 1);

    pthread_t employees[NUM_EMPLOYEES];

    // 创建员工线程
    for (long i = 1; i <= NUM_EMPLOYEES; i++) {
        pthread_create(&employees[i-1], NULL, employee, (void*)i);
    }

    // 等待所有员工完成
    for (int i = 0; i < NUM_EMPLOYEES; i++) {
        pthread_join(employees[i], NULL);
    }

    // 销毁信号量
    sem_destroy(&mutex);
    sem_destroy(&all_here);
    sem_destroy(&entry_mutex);

    return 0;
}
```

### 题目

搜索-插入-删除问题。三个线程对一个单链表进行并发的访问，分别进行搜索、插入和删除。搜索线程仅仅读取链表，因此多个搜索线程可以并发。插入线程把数据项插入到链表最后的位置；多个插入线程必须互斥防止同时执行插入操作。但是，一个插入线程可以和多个搜索线程并发执行。最后，删除线程可以从链表中任何一个位置删除数据。一次只能有一个删除线程执行；删除线程之间，删除线程和搜索线程，删除线程和插入线程都不能同时执行。

请编写三类线程的同步互斥代码，描述这种三路的分类互斥问题。

### 答案

**答案：**

```C++
#include <pthread.h>
#include <semaphore.h>
#include <stdio.h>

// 全局同步变量
sem_t insert_mutex;    // 插入互斥信号量（初始值1）
sem_t delete_mutex;    // 删除互斥信号量（初始值1）
pthread_mutex_t search_count_mutex = PTHREAD_MUTEX_INITIALIZER;
pthread_mutex_t sync_mutex = PTHREAD_MUTEX_INITIALIZER;
pthread_cond_t no_search = PTHREAD_COND_INITIALIZER;

int search_count = 0;  // 当前正在搜索的线程数
int delete_waiting = 0;// 等待删除的线程数

// 搜索线程函数
void* search_thread(void* arg) {
    // 获取同步锁
    pthread_mutex_lock(&sync_mutex);
    // 等待删除线程完成
    while (delete_waiting > 0) {
        pthread_cond_wait(&no_search, &sync_mutex);
    }
    pthread_mutex_unlock(&sync_mutex);

    // 增加搜索计数
    pthread_mutex_lock(&search_count_mutex);
    search_count++;
    pthread_mutex_unlock(&search_count_mutex);

    // 执行搜索操作（安全地与插入线程并发）
    printf("搜索线程 %ld: 正在搜索...\n", (long)arg);

    // 减少搜索计数
    pthread_mutex_lock(&search_count_mutex);
    search_count--;
    // 如果删除线程在等待且没有搜索线程了
    if (search_count == 0 && delete_waiting > 0) {
        pthread_cond_signal(&no_search);
    }
    pthread_mutex_unlock(&search_count_mutex);

    return NULL;
}

// 插入线程函数
void* insert_thread(void* arg) {
    // 1. 插入线程互斥
    sem_wait(&insert_mutex);

    // 获取同步锁
    pthread_mutex_lock(&sync_mutex);
    // 等待删除线程完成
    while (delete_waiting > 0) {
        pthread_cond_wait(&no_search, &sync_mutex);
    }
    pthread_mutex_unlock(&sync_mutex);

    // 执行插入操作（可与搜索线程并发）
    printf("插入线程 %ld: 正在插入...\n", (long)arg);

    // 释放插入锁
    sem_post(&insert_mutex);
    return NULL;
}

// 删除线程函数
void* delete_thread(void* arg) {
    // 1. 删除线程互斥
    sem_wait(&delete_mutex);

    // 声明等待删除
    pthread_mutex_lock(&sync_mutex);
    delete_waiting++;
    // 等待所有搜索和插入完成
    while (search_count > 0) {
        pthread_cond_wait(&no_search, &sync_mutex);
    }
    pthread_mutex_unlock(&sync_mutex);

    // 执行删除操作（完全互斥）
    printf("删除线程 %ld: 正在删除...\n", (long)arg);

    // 完成删除
    pthread_mutex_lock(&sync_mutex);
    delete_waiting--;
    // 唤醒所有等待线程
    pthread_cond_broadcast(&no_search);
    pthread_mutex_unlock(&sync_mutex);

    // 释放删除锁
    sem_post(&delete_mutex);
    return NULL;
}

int main() {
    // 初始化信号量
    sem_init(&insert_mutex, 0, 1);
    sem_init(&delete_mutex, 0, 1);

    // 创建线程
    pthread_t threads[9]; // 3搜索 + 3插入 + 3删除
    long thread_id = 0;

    // 搜索线程
    for (int i = 0; i < 3; i++) {
        pthread_create(&threads[thread_id++], NULL, search_thread, (void*)thread_id);
    }

    // 插入线程
    for (int i = 0; i < 3; i++) {
        pthread_create(&threads[thread_id++], NULL, insert_thread, (void*)thread_id);
    }

    // 删除线程
    for (int i = 0; i < 3; i++) {
        pthread_create(&threads[thread_id++], NULL, delete_thread, (void*)thread_id);
    }

    // 等待所有线程结束
    for (int i = 0; i < 9; i++) {
        pthread_join(threads[i], NULL);
    }

    // 销毁资源
    sem_destroy(&insert_mutex);
    sem_destroy(&delete_mutex);
    pthread_mutex_destroy(&search_count_mutex);
    pthread_mutex_destroy(&sync_mutex);
    pthread_cond_destroy(&no_search);

    return 0;
}
```
