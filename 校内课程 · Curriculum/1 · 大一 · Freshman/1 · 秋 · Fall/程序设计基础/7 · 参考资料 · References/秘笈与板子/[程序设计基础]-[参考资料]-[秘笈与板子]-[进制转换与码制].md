## 任意进制转十进制

```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include <math.h>

int stringToDecimal(const char *str, int base) {
    int decimal = 0;
    int len = strlen(str);
    int i;

    // 从右往左遍历字符串中的每个字符
    for (i = 0; i < len; ++i) {
        char c = tolower(str[len - 1 - i]); // 转换为小写以便统一处理
        int digitVal = 0;

        // 检查字符是数字还是字母，并计算其代表的数值
        if (isdigit(c)) {
            digitVal = c - '0';
        } else if (isalpha(c)) {
            digitVal = c - 'a' + 10; // 假设是小写字母，计算其代表的十六进制数值
        } else {
            // 打印错误消息并返回-1表示转换失败
            printf("Invalid character in the input string.\n");
            return -1;
        }

        // 检查数值是否在允许的范围内
        if (digitVal >= base) {
            // 打印错误消息并返回-1表示转换失败
            printf("Invalid digit for the given base.\n");
            return -1;
        }

        // 根据位权累加计算十进制数值
        decimal += digitVal * (int)pow(base, i);
    }

    return decimal; // 返回转换后的十进制数
}

int main() {
    char inputStr[65]; // 假设输入字符串不会超过64个字符
    int base;
    int decimalNumber;

    printf("Enter a number string: ");
    scanf("%64s", inputStr); // 读取字符串，限制最大长度为64
    printf("Enter the base of the number string: ");
    scanf("%d", &base);

    if (base < 2 || base > 16) {
        printf("Invalid base. Base should be between 2 and 16.\n");
        return 1;
    }

    decimalNumber = stringToDecimal(inputStr, base);
    if (decimalNumber != -1) {
        printf("The decimal representation of the number is: %d\n", decimalNumber);
    } else {
        printf("Failed to convert the number string to decimal.\n");
    }

    return 0;
}
```

## 十进制转任意进制

```c
#include <stdio.h>  
#include <string.h>  
#include <math.h>  
  
void decimalToBaseN(int decimal, int base, char result[]) {  
    int index = 0;  
    int tempDecimal = decimal;  
      
    // 确定结果字符串的长度  
    do {  
        tempDecimal /= base;  
        index++;  
    } while (tempDecimal > 0);  
      
    // 初始化结果字符串为全0，用于后续的反向填充  
    memset(result, '0', index);  
    result[index] = '\0'; // 字符串结束符  
      
    // 将十进制数转换为指定进制的数，并填充到结果字符串中  
    tempDecimal = decimal;  
    while (tempDecimal > 0) {  
        int digit = tempDecimal % base;  
        if (digit < 10) {  
            result[--index] = '0' + digit; // 数字0-9  
        } else {  
            result[--index] = 'A' + (digit - 10); // 字母A-F（对于16进制）  
        }  
        tempDecimal /= base;  
    }  
}  
  
int main() {  
    int decimalNumber;  
    int base;  
    char result[33]; // 假设结果不会超过32个字符（包括结束符）  
  
    printf("Enter a decimal number: ");  
    scanf("%d", &decimalNumber);  
    printf("Enter the base to convert to (2-16): ");  
    scanf("%d", &base);  
  
    if (base < 2 || base > 16) {  
        printf("Invalid base. Base should be between 2 and 16.\n");  
        return 1;  
    }  
  
    decimalToBaseN(decimalNumber, base, result);  
    printf("The number in base %d is: %s\n", base, result);  
  
    return 0;  
}
```

## 输出32位原码反码补码

```c
有符号整型变量（如 int）在计算机中以补码的形式存储。
正数的原码、反码、补码均相同。
负数 n的原码与其绝对值 |n|=−n的原码除符号位以外均相同；
负数 n的反码等于对 |n|的原码取反，n的反码与 n原码相比除符号位以外均不同；
负数 n的补码代表的二进制数等于其反码代表的二进制数加一。
#include <stdio.h>

int main() {
    int num;
    while(scanf("%d", &num)!=EOF) {
        unsigned int unsigned_num = (unsigned int) num;
        unsigned int original, reversed, complement;

        if (num >= 0) {
            original = unsigned_num;
            reversed = unsigned_num;
            complement = unsigned_num;
        }
        else {
            original = (1 << 31) | (unsigned int) (-num);//原码
            reversed = original ^ 0xFFFFFFFF;//和32位全是1的取异或全部取反
            reversed |= (1 << 31);//反码
            complement = reversed + 1;//补码=原码+1
        }

        for (int i = 31; i >= 0; i--) {
            printf("%u", (original >> i) & 1);
        }
        printf("\n");
        for (int i = 31; i >= 0; i--) {
            printf("%u", (reversed >> i) & 1);
        }
        printf("\n");
        for (int i = 31; i >= 0; i--) {
            printf("%u", (complement >> i) & 1);
        }
        printf("\n");
    }
    return 0;
}
```

## 输出补码

```c
#include <stdio.h>

// 函数用于打印一个无符号整数的二进制表示
void print_binary(unsigned int num, int bits) {
    for (int i = bits - 1; i >= 0; i--) {
        printf("%u", (num >> i) & 1);
    }
}

int main() {
    int num;
    printf("请输入一个整数: ");
    scanf("%d", &num);
    printf("补码（也是内存中的表示）: ");
    print_binary((unsigned int)num, 32); // 强制转换为无符号整数以避免符号扩展
    printf("\n");

    return 0;
}
```

```c
{
    int decimalNumber = 0, i = 0, remainder;
```
