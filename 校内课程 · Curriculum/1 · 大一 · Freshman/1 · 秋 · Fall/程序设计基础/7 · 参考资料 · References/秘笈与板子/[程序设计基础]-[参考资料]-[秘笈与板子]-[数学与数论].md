// 简化
int gcd(int a, int b) {
    return (b==0) ? a : gcd(b, a % b);

## 一元二次方程求根

```c
#include <stdio.h>
#include <math.h>
int main(){
    double a, b, c, delta, r1, r2;
    double eps = 1e-9; //浮点数的精度误差
    scanf("%lf%lf%lf",&a,&b,&c);
    if(a==0){ //a为0，不是一元二次方程
        if(b==0){ //a为0的同时，b也为0，不是方程
            if(c==0) printf("infinite solutions");
            else printf("No real root");
        }else{ //a为0的同时，b不为0，则是一元一次方程
            printf("%.2f",-c/b);
        }
    }else{ //a不为0，是一元二次方程
        delta = b * b - 4 * a * c;
        if(fabs(delta)<=eps){//delta值趋近0，一个根
            r1 = r2 = -b / (2 * a);
            printf("%.2f",r1);
        }else if(delta>0){ //delta值大于0，两个根
            r1 = (-b + sqrt(delta)) / (2 * a);
            r2 = (-b - sqrt(delta)) / (2 * a);
            if(r1<r2) printf("%.2f %.2f",r1,r2);
            else printf("%.2f %.2f",r2,r1);
        }else{ //delta值小于0，无实根
            printf("No real root");
        }
    } 
    return 0;
}
```

## 判断是否是递减数

```c
#include <stdio.h>
#include <stdbool.h>

bool function(int num) {
    int prev_digit = -1;/*在is_strictly_increasing函数中，判断前一位数字prev_digit是否是-1的原因是为了处理输入整数的最高位数字。因为对于最高位数字，它并没有“前一位”数字与之比较，所以我们需要在第一次比较之前跳过这个检查。

将prev_digit初始化为-1是一个技巧，它允许我们在循环的第一次迭代中避免进行比较。这是因为-1作为一个哨兵值，在逻辑上总是小于任何有效的数字位（0到9）。所以，当prev_digit为-1时，我们知道还没有前一位数字与当前数字进行比较，因此可以安全地跳过比较，并设置prev_digit为当前数字。

从第二次迭代开始，prev_digit将包含一个有效的数字，此时我们就可以开始比较它是否小于当前数字了。*/
    while (num > 0) {
        int current_digit = num % 10;
        if (prev_digit != -1 && prev_digit >= current_digit) {
            return false;
        }
        prev_digit = current_digit;
        num /= 10;
    }
    return true;
}

int main() {
    int n;
    while(scanf("%d", &n)!=EOF) {
        if (function(n))
            printf("Yes\n");
        else
            printf("No\n");
    }
    return 0;
}
```

快速幂
/*快速求 a 的 b 次方模 p 的结果. 时间复杂度为 O(logN)*/
long long qpow(long long a, unsigned long long b, long long p) {
    long long ans = 1;
    a = a % p;
    while(b) {
        if(b & 1)
            ans = (ans * a) % p;
        b >>= 1;
        a = a * a % p;
    }
    return ans;
}

## 解一元二次方程

```c
#include <stdio.h>
#include <math.h>
#define eps 1e-9
int main()
{
    double a,b,c;
    scanf("%lf%lf%lf",&a,&b,&c);
    double delta = b * b - 4 * a * c;
// a = 0，b = 0 时
    if(fabs(a) < eps && fabs(b) < eps){
        if(fabs(c) < eps)
            printf("infinite solutions\n");
        else
            printf("No real root");
    }
// a = 0, b != 0 时
    else if(fabs(a) < eps)
        printf("%.2f\n",-c/b);
// a != 0 时，即二元一次方程的讨论
    else if(delta < 0)
        printf("No real root\n");
    else if(fabs(delta) < eps)
        printf("%.2f\n",-b/(2*a));
    else {
        double x1 = (-b + sqrt(delta)) / (2 * a);
        double x2 = (-b - sqrt(delta)) / (2 * a);
        if(x1 < x2)
            printf("%.2f %.2f\n",x1,x2);
        else
            printf("%.2f %.2f\n",x2,x1);
    }
    return 0;
}
```

## 计算两个向量的夹角

```c
#include <stdio.h>
#include <math.h>
#define PI 3.1415926538
int main(){
    double a[3], b[3], c[3];
    double dotx, crossx, theta;
    scanf("%lf%lf%lf", &a[0], &a[1], &a[2]);
    scanf("%lf%lf%lf", &b[0], &b[1], &b[2]);
    //求a、b的叉乘，保存在c中
    c[0] = a[1] * b[2] - a[2] * b[1];
    c[1] = a[2] * b[0] - a[0] * b[2];
    c[2] = a[0] * b[1] - a[1] * b[0];
    //求叉乘结果c的模长
    crossx = sqrt(c[0] * c[0] + c[1] * c[1] + c[2] * c[2]);
    //求a、b的点积
    dotx = a[0] * b[0] + a[1] * b[1] + a[2] * b[2];
    //求atan2夹角，结果为弧度，转换为角度
    theta = atan2(crossx, dotx) * 180.0 / PI;
    printf("angle = %0.2f\n", theta);
    return 0;
}
```
