## 大V图案

```c
#include <stdio.h>
int main(){
    int n, i, j;
    scanf("%d", &n);
    //循环n次
    for(i = n; i >= 1; i--){
        // 第i行，输出n-i个空格
        for(j = 1; j <= n-i; j++) putchar('');
        putchar('\\'); // 输出反斜杠
        // 第i行，输出2i-2个空格
        for(j = 1; j <= 2 * i-2; j++) putchar('');
        putchar('/'); // 输出斜杠
        putchar('\n');
    }
}
```

## 打印平均分和高出平均分的成绩

```c
#include <stdio.h>
int main(){
    int score[201], sum=0;//成绩数组和总分数，数组的长度为201(200即可，但可适当长一些）
    int n = 0, i; //学生数量n（初始化为0）和临时变量i
    double average; //平均分，double类型
    scanf("%d", &score[n]);//第一个成绩输入到数据的第一个元素，n的初值为0
    while(score[n] != -1){ //判断成绩是否-1 
        sum += score[n]; //累加成绩
        n++; //人数+1
        scanf("%d", &score[n]);//输入下一个成绩
    }
    if (n>0){ //输入的有效成绩不为0
        average = (double)sum/n; //计算平均分，注意理解如何求double类型的平均分（后面讲解）
        printf("%d/%d = %.2f\n", sum, n, average);//打印平均分，double打印格式（后面讲解）
        i = 0; //初始化i为0，从第0个开始依次访问数组中的每个元素
        while(i<n){ //依次访问成绩数组，打印出高于平均分的序号和成绩
            if(score[i]>average)//如果分数高于平均分，打印输出
                printf("%d : %d\n", i+1, score[i]);//注意输出i+1，数组下标从0开始
            i++;
        }
    }
    return 0;
}
```

```c
/*定义字符数组str1, str2, 字符 c*/

/*返回字符串 str1 的长度, 从所给指针的地址开始直到遇见第一个'\0'. 注意, 该函数为线性时间复杂度, 请勿在循环条件判断中随意加入使用!!!*/
strlen(str1);
/*按照字典序比较 str1 和 str2, 若 str1 较小则返回负值, 较大则返回正值, 相同则返回 0. 注意返回的值是不确定的, 只有符号确定!*/
strcmp(str1, str2);
/*将 str2 中的字符复制到 str1 中, 原 str1 的内容会被覆盖. 注意 str1 的大小需要装得下 str2 的内容!*/
strcpy(str1, str2);
/*将 str2 接到 str1 的末尾, 覆盖原 str1 末尾的 '\0'. 注意 str1 的大小需要装得下原 str1 和 str2 的内容, 且 str1 和 str2 不能重叠!*/
strcat(str1, str2);
/*在 str1 中查找 str2. 若 str2 为 str1 的子串, 则返回 str2 在 str1 中首次出现的地址, 否则返回 NULL*/
strstr(str1, str2);
/*在 str1 中查找字符 c. 若找到则返回 c 在 str1 中首次出现的地址, 否则返回 NULL*/
strchr(str1, c);
/*在 str1 中反向查找字符 c. 若找到则返回 c 在 str1 中最后一次出现的地址, 否则返回 NULL*/
strrchr(str1, c);

strlen 函数的使用注意
再次强调一下 strlen 函数的使用. 这一函数的实现本身就是 $O(n)$ 的, 如果把它放到循环条件判断中, 每次都要重新计算一次, 复杂度就变为 $O(n^2)$ 了, 如下:
/*禁忌写法*/
for(i = 0; i < strlen(str); ++i) {
    /*TO DO*/
}

这也是有些题目莫名其妙 TLE 的原因. 正确写法如下:
/*正确写法*/
int len = strlen(str);
for(i = 0; i < len; ++i) {
    /*TO DO*/
}
```

## 数字变字符（压缩）

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main() 
{
    int n;
    scanf("%d", &n);
    
    int arr[n];
    for (int i = 0; i < n; i++) 
    {
        scanf("%d", &arr[i]);
    }
    
    char compressed[n * 2]; // 压缩结果字符串，初始化为两倍于输入数组长度
    int compressedIndex = 0; // 压缩结果字符串当前索引
    
    for (int i = 0; i < n; i++) 
    {
        int num = arr[i];
        
        if (num >= 10 && num <= 35) 
        {
            compressed[compressedIndex] = 'A' + num - 10;
        }
        else if (num >= 0 && num <= 9) 
        {
            compressed[compressedIndex] = '0' + num;
        }
        else if (num >= -10 && num <= -1)
        {
            compressed[compressedIndex] = "!@#$%^&*()"[-num-1];
        }
        else if (num >= -36 && num <= -11)
        {
            compressed[compressedIndex] = 'a' - num - 11;
        }
        compressedIndex++;
    }
    compressed[compressedIndex] = '\0'; // 压缩结果字符串末尾加上结束符
    
    printf("%s", compressed);
    return 0;
}
```

## 自定义格式串

```c
#include <stdio.h>
#include <string.h>

int main() {
    char str[1005];
    char s[1005];
    char format[105];
    char format_p[105];
    int k, x;
    fgets(str, sizeof(str), stdin); 
    str[strcspn(str, "\n")] = 0; 
    fgets(format, sizeof(format), stdin); 
    format[strcspn(format, "\n")] = 0; 
    scanf("%d%d", &k, &x);
    sscanf(str, format, s);
    sprintf(format_p, "%%%d.%ds", k, x);
    printf(format_p, s);
    return 0;
}
```

## 输入一组数求最大值

```c
int main() {  
    int n, i, num, max;  
    scanf("%d", &n);  //请输入数字的个数
    int arr[n]; // 声明一个大小为n的数组来存储数字   
    for (i = 0; i < n; i++) {  
        scanf("%d", &arr[i]); // 从用户那里读取数字并存储在数组中  
    }  
  
    max = arr[0]; // 假设第一个数字是最大的  
    for (i = 1; i < n; i++) {  
        if (arr[i] > max) max = arr[i]; // 如果找到更大的数字，则更新最大值  
         }  
    printf("%d\n", max);  
    return 0;  
}
```
