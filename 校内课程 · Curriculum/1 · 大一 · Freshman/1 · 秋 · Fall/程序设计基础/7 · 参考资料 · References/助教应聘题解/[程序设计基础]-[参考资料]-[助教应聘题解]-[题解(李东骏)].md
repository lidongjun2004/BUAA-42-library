# `E` 前20长的字符串

| 难度 | 考点 |
| --- | --- |
| 4 | 字符串、排序 |

## 题目分析

第一行强调空间要求4096kb，非常小。所以数组规模不能开很大。又根据数据范围n的上限为100000。因此不可能记录下全部n个字符串。思考题目需要我们做什么，需要求出前20个最长字符串。所以至少需要记录20个最长长度为1000的字符串。这么大的数组规模与题目空间要求基本相符。
下面考虑如何构造。
考虑维护一个20个字符串的由长到短排列，每新输入进来一个字符串，与最短相比，若新字符串更长，则替换最短字符串，并维持数组有序。否则继续输入下一个字符串。
由于题目中要求按照输入顺序输出，故考虑再维护一个变量no，记录每个字符串输入的顺序。输入过程中，维护数组先按照字符串长度排序，其次按照no来排序。
在输入完成后，按照no重新排序，按顺序输出即可。

### 示例代码

```c
#include<stdio.h>
#include<math.h>
#include<string.h>
#include<stdlib.h>
typedef struct node
{
 char str[1005];//记录字符串 
 int no,len;//分别记录输入顺序和字符串长度 
}word;
word list[20];//维护前20长字符串的数组 
char s[1005];
int n;//记录已输入字符串数 
void insert(char *str,int len,int no)
{
 int rank=0;
 for(int i=18;i>=0;i--)//找到应该插入的位置 
 {
  if(len>list[i].len)
  {
   list[i+1].len=list[i].len;
   list[i+1].no=list[i].no;
   strcpy(list[i+1].str,list[i].str);
  }
  else 
        {
            rank=i+1;
            break;
        }
 }
 list[rank].len=len;
 strcpy(list[rank].str,str);
 list[rank].no=n;
}
int cmp(const void *a,const void *b)//先从长到短，长度相同从前往后 
{
 word x=*(word*)a,y=*(word*)b;
 if(x.len-y.len)return y.len-x.len;
 return x.no-y.no;
}
int cmp2(const void *a,const void *b)//按先后顺序排序 
{
 word x=*(word*)a,y=*(word*)b;
 return x.no-y.no;
}
int main()
{
 for(int i=0;i<20;i++)//先输入20个字符串，填充数组 
 {
  gets(s);
  strcpy(list[i].str,s);
  list[i].no=++n;
  list[i].len=strlen(s);
 }
 qsort(list,20,sizeof(word),cmp);//将数组按照从长到短排序 
    while(gets(s)!=NULL)
    {
     int len=strlen(s);
     n++;
     if(len>list[19].len)insert(s,len,list[19].no);//当前字符串长度大于数组中最短，执行插入操作 
 }
 qsort(list,20,sizeof(word),cmp2);
 for(int i=0;i<20;i++)printf("%s\n",list[i].str);
 return 0;
}
```
