## A 美味排序

| 难度 | 考点 |
| --- | --- |
| 1 | 排序 |

### 题目

#### 题目描述

读入 $n$ 个实数，把它们从小到大排序后输出，每个数保留 4 位小数。

#### 输入格式

第一行一个整数 $n$ 。接下来 $n$ 个实数，用空格或换行分隔均可。

#### 输出格式

一行 $n$ 个实数，按从小到大排列，每个数保留小数点后 4 位，相邻两数之间用一个空格分隔。

#### 样例输入

```text
5
3.5 1.25 -2 0 10
```

#### 样例输出

```text
-2.0000 0.0000 1.2500 3.5000 10.0000
```

#### 数据范围

$n \le 2000$

### 答案

### 题目分析

我们给出五种排序方式，同学们可以自己学习。

### 示例代码 1

```c
/* 冒泡排序，在课上已经介绍过。
 * 时间复杂度 O(n^2)
 * 参考资料：https://oi-wiki.org/basic/bubble-sort/
 */
#include <stdio.h>

double arr[2010];

void bubbleSort(double arr[], int n) {
    for (int i = 0; i < n; i++) {
        for (int j = i + 1; j < n; j++) {
            if (arr[i] > arr[j]) {
                double temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;
            }
        }
    }
}

int main() {
    int n = 0;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        scanf("%lf", &arr[i]);
    }
    bubbleSort(arr, n);
    for (int i = 0; i < n; i++) {
        printf("%.4lf ", arr[i]);
    }
    return 0;
}
```

### 示例代码 2

```c
/* 插入排序，把数组划分为有序的前一部分和无序的后一部分，每次从后一部分取一个元素，插入到前一部分的正确位置当中。
 * 时间复杂度 O(n^2)
 * 参考资料：https://oi-wiki.org/basic/insertion-sort/
 */
#include <stdio.h>

double arr[2010];

void insertSort(double arr[], int n) {
    for (int i = 1; i < n; i++) {
        double key = arr[i];
        int j = i - 1;
        while (j >= 0 && arr[j] > key) {
            arr[j + 1] = arr[j];
            --j;
        }
        arr[j + 1] = key;
    }
}

int main() {
    int n = 0;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        scanf("%lf", &arr[i]);
    }
    insertSort(arr, n);
    for (int i = 0; i < n; i++) {
        printf("%.4lf ", arr[i]);
    }
    return 0;
}
```

### 示例代码 3

```c
/* 选择排序，每次从数组的待排序部分中选出最小的元素，然后放到待排序部分最前面来
 * 时间复杂度 O(n^2)
 * 参考资料：https://oi-wiki.org/basic/selection-sort/
 */
#include <stdio.h>

double arr[2010];

void selectSort(double arr[], int n) {
    for (int i = 0; i < n - 1; i++) {
        int minIndex = i;
        for (int j = i + 1; j < n; j++) {
            if (arr[j] < arr[minIndex]) {
                minIndex = j;
            }
        }
        if (minIndex != i) {
            double temp = arr[i];
            arr[i] = arr[minIndex];
            arr[minIndex] = temp;
        }
    }
}

int main() {
    int n = 0;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        scanf("%lf", &arr[i]);
    }
    selectSort(arr, n);
    for (int i = 0; i < n; i++) {
        printf("%.4lf ", arr[i]);
    }
    return 0;
}
```

### 示例代码 4

```c
/* 归并排序，借用 E4-F 中合并两个有序数组的函数 merge()，
 * 每次把待排序数组从中间划分为两半，然后递归地排序左一半和右一半，再利用 E4-F 中线性时间内合并两个有序数组的算法加以合并，
 * 如果划分到只有一个元素，就直接返回。
 * 时间复杂度 O(nlogn)
 * 参考资料：https://oi-wiki.org/basic/merge-sort/
 */
#include <stdio.h>

double arr[2010];
double temp[2010];

void merge(double arr[], int left, int mid, int right) {
    int i = left, j = mid + 1;
    int k = 0;
    while (i <= mid && j <= right) {
        if (arr[i] <= arr[j]) {
            temp[k++] = arr[i++];
        } else {
            temp[k++] = arr[j++];
        }
    }

    while (i <= mid) {
        temp[k++] = arr[i++];
    }

    while (j <= right) {
        temp[k++] = arr[j++];
    }

    for (i = left, k = 0; i <= right; i++, k++) {
        arr[i] = temp[k];
    }
}

void mergeSort(double arr[], int n) {
    if (n < 2) {
        return;
    }
    int mid = n / 2;
    mergeSort(arr, mid);
    mergeSort(arr + mid, n - mid);
    merge(arr, 0, mid - 1, n - 1);
}

int main() {
    int n = 0;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        scanf("%lf", &arr[i]);
    }
    mergeSort(arr, n);
    for (int i = 0; i < n; i++) {
        printf("%.4lf ", arr[i]);
    }
    return 0;
}
```

/* 快速排序，直接调用 stdlib.h 中现成的 qsort() 函数
 * 时间复杂度 O(nlogn)
 * 参考资料：https://oi-wiki.org/basic/quick-sort/

#include <stdio.h>
#include <stdlib.h>

double arr[2010];

int compare(const void* a, const void* b) {
    double num_a = *(double*)a;
    double num_b = *(double*)b;
    if (num_a < num_b) {
        return -1;
    } else if (num_a > num_b) {
        return 1;
    } else {
        return 0;
    }
}

    int n = 0;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        scanf("%lf", &arr[i]);
    }

    for (int i = 0; i < n; i++) {
        printf("%.4lf ", arr[i]);
    }
    return 0;
}

## B 数组重排

| 难度 | 考点 |
| --- | --- |
| 2 | 数组 |

### 题目

#### 题目描述

多组数据。每组给出 $n$ 个整数，判断能否重新排列这 $n$ 个数，使得任意相邻两个数模 3 的余数都不相同。

#### 输入格式

第一行一个整数 $t$ ，表示数据组数。每组数据第一行一个整数 $n$ ，第二行 $n$ 个整数，用空格分隔。

#### 输出格式

每组输出一行：可以做到输出 `YES` ，否则输出 `NO` 。

#### 样例输入

```text
2
4
1 2 3 4
3
0 0 0
```

#### 样例输出

```text
YES
NO
```

### 答案

### 题目分析

首先注意到元素的具体值、初始位置都是不重要的。与答案有关的只有模 $3$ 为 $0$， $1$， $2$ 的元素个数。

记模 $3$ 为 $0$ 的元素个数为 $c_0$，同理定义 $c_1$ 和 $c_2$。

下证答案为 NO 当且仅当 $c_{\max} > \left\lceil \dfrac{n}{2} \right\rceil$.

首先证明： $c_{\max} > \left\lceil \dfrac{n}{2} \right\rceil$ 时答案为 NO。这个其实是显然的。因为两两间隔放至多放 $\left\lceil \dfrac{n}{2} \right\rceil$ 个。

然后证明： $c_{\max} \le \left\lceil \dfrac{n}{2} \right\rceil$ 时答案为 YES。

此处不妨设 $c_0 > c_1 > c_2$，因为性质与具体值无关。

我们将 $c_0$ 个 $0$ 放在 $1, 3, 5..$ 等奇数位上。剩下 $\left\lceil \dfrac{n}{2} \right\rceil - c_0$ 个奇数位。

接着我们将 $c_1$ 放满剩下的奇数位。这一定是可以做到的，因为如果 $c_1 < \left\lceil \dfrac{n}{2} \right\rceil - c_0$，那么就有 $c_2 > c_0$ 了，与题设矛盾。

放满后只剩偶数位。我们向 $2, 4, 6...$ 等偶数位上放剩下的 $1$。可以证明此时放下的 $1$ 都是被 $0$ 包围的。否则又有 $c_1 > c_0$。

最后剩下的位置全部放 $2$，这些位置都是被 $0$ 或 $1$ 隔开的，符合要求。

以上为一个合法的构造方案，证明了 $c_{\max} \le \left\lceil \dfrac{n}{2} \right\rceil$ 时答案为 YES。

注意特判一下 $n = 1$ 的情景。

### 示例代码

```c
#include <stdio.h>

int t, n;
int a[1005];

int solve() {
    scanf("%d", &n);
    int cnt0 = 0, cnt1 = 0, cnt2 = 0;
    for (int i = 1; i <= n; i++) {
        scanf("%d", &a[i]);
        if (a[i] % 3 == 0) cnt0 += 1;
        if (a[i] % 3 == 1) cnt1 += 1;
        if (a[i] % 3 == 2) cnt2 += 1;
    }
    if (n == 1) return 1;
    if (cnt0 > (n + 1) / 2 || cnt1 > (n + 1) / 2 || cnt2 > (n + 1) / 2) {
        return 0;
    } else {
        return 1;
    }
}

int main() {
    scanf("%d", &t);
    while (t--) {
        if (solve()) puts("YES");
        else puts("NO");
    }
    return 0;
}
```

## C 约瑟夫

### 题目

#### 题目描述

$n$ 个人围成一圈，编号从 1 到 $n$ 。从编号 1 的人开始报数，每数到第 $m$ 个人就让他出圈，再从下一个人继续报数。按出圈的先后顺序输出所有人的编号。

#### 输入格式

一行两个整数 $n$ 和 $m$ ，用空格分隔。

#### 输出格式

一行 $n$ 个整数，表示出圈顺序，相邻两数之间用一个空格分隔。

#### 样例输入

```text
5 3
```

#### 样例输出

```text
3 1 5 2 4
```

#### 数据范围

$n \le 100$

### 答案

### 题目分析

这个经典的问题由约瑟夫于公元 $1$ 世纪提出，尽管他当时只考虑了 $k = 2$ 的情况。现在我们可以用许多高效的算法解决这个问题。

最朴素的算法最朴素的算法莫过于直接枚举。通过使用一个数组记录当前人的状态，然后从前往后依次枚举即可。我们的题目 $n \le 100$，该方法完全能够解决。

同时，值得注意的是，在下列代码的实现过程中， $j$ 的循环处理本质上是在使用数组模拟链表，同学们将在下学期的数据结构课程中详细学习相关知识。同学们也可以尝试使用指针来真正实现链表

### 示例代码

```c
#include<stdio.h>
int a[101];
int main ( )
{
    int n,m;
    scanf("%d%d",&n,&m);
    int j=0;
    for(int i=1;i<=n;i++)
    {
        int p=0;
        while(p!=m)
        {
            j++;
            if(j>n)
            j-=n;
            if(a[j]==0)
            p++;
        }
        a[j]=1;
        printf("%d ",j);
    }
    return 0 ;
}
```

大家可以注意到，该方法其实并不优秀，复杂度达到了 $\Theta(n^2)$ 那有没有什么更优秀的方法呢？当然是有的，传送门如下：

## D 冰雪国都

| 难度 | 考点 |
| --- | --- |
| 3 | 字符串，进制转换 |

### 题目

#### 题目描述

冰雪国都用一种特殊的记数法表示整数：一个字符串的最后一位表示进制，它前面的各位是该进制下的数位，其中字符 `0` 到 `9` 表示数值 0 到 9，字符 `a` 到 `z` 表示数值 10 到 35。给定若干个这样的字符串，求它们对应的十进制数值。

#### 输入格式

第一行一个整数 $t$ 。接下来 $t$ 行，每行一个字符串。

#### 输出格式

$t$ 行，每行一个整数，表示对应字符串表示的十进制数值。

#### 样例输入

```text
2
1012
ffg
```

#### 样例输出

```text
5
255
```

### 答案

### 题目分析

该题考察大家对于字符串相关函数的运用，并顺道复习进制转化。

你可能需要用到

```c
strlen(const char *str)
```

通过该函数，可以得到字符串的长度，从而能够读取到通过 `scanf("%s")` 或 `gets` 得到字符串的最后一位，从而得到原始进制数。

后续就是简单的进制转化了，由于数据范围并不大，你可以不使用高精度。

代码如下：

```c
#include<stdio.h>
#include<string.h>
char ch[100001];
long long a[100001];
int main ( )
{
    int t;
    scanf("%d\n",&t);
    while(t--)
    {
        gets(ch);
        int l=strlen(ch);
        for(int i=0;i<l;i++)
        {
            if(ch[i]>='0'&&ch[i]<='9')
            a[i]=ch[i]-'0';
            else
            a[i]=ch[i]-'a'+10;
        }
        long long ans=0,p=1;
        for(int i=l-2;i>=0;i--)
        {
            ans+=p*a[i];
            p*=a[l-1];
        }
        printf("%lld\n",ans);
    }
    return 0 ;
}
```

## E HYPER MEMORIES

| 难度 | 考点 |
| --- | --- |
| 3 | 二维数组、前缀和 |

### 题目

#### 题目描述

给定一个 $n$ 行 $m$ 列的整数矩阵和 $q$ 次询问，每次询问给出一个子矩阵的左上角与右下角坐标，求该子矩阵内所有元素之和。

#### 输入格式

第一行两个整数 $n$ 和 $m$ 。接下来 $n$ 行，每行 $m$ 个整数，表示矩阵。接下来一行一个整数 $q$ 。接下来 $q$ 行，每行四个整数 $i_1$ 、 $j_1$ 、 $i_2$ 、 $j_2$ ，表示子矩阵左上角是第 $i_1$ 行第 $j_1$ 列、右下角是第 $i_2$ 行第 $j_2$ 列。

#### 输出格式

$q$ 行，每行一个整数，表示对应子矩阵内所有元素之和。

#### 样例输入

```text
3 3
1 2 3
4 5 6
7 8 9
2
1 1 2 2
2 2 3 3
```

#### 样例输出

```text
12
28
```

#### 数据范围

$n, m \le 2000$ ，元素和需要用 `long long` 存储。建议用二维前缀和。

### 答案

### 题解

其实，题解已经放在题面下面的提示了。不过如果你不想看图片的话，我们有更清晰简洁的公式！

定义 $S_{i,j} = \sum_{i'=1}^{i} \sum_{j'=1}^{j} a_{i',j'}$，有

$$
\begin{aligned}
\sum_{i=i_1}^{i_2} \sum_{j=j_1}^{j_2} a_{i,j} &= \sum_{i=i_1}^{i_2} \sum_{j=j_1}^{j_2} a_{i,j} \\
&= \sum_{i=i_1}^{i_2} \left( \sum_{j=1}^{j_2} a_{i,j} - \sum_{j=1}^{j_1-1} a_{i,j} \right) - \sum_{i=1}^{i_1-1} \left( \sum_{j=1}^{j_2} a_{i,j} - \sum_{j=1}^{j_1-1} a_{i,j} \right) \\
&= \sum_{i=1}^{i_2} \sum_{j=1}^{j_2} a_{i,j} - \sum_{i=1}^{i_2} \sum_{j=1}^{j_1-1} a_{i,j} - \sum_{i=1}^{i_1-1} \sum_{j=1}^{j_2} a_{i,j} + \sum_{i=1}^{i_1-1} \sum_{j=1}^{j_1-1} a_{i,j} \\
&= S_{i_2,j_2} - S_{i_2,j_1-1} - S_{i_1-1,j_2} + S_{i_1-1,j_1-1}
\end{aligned}
$$

而 $S_{i,j}$ 是好求的：

$$
\begin{aligned}
S_{i,j} &= \sum_{i'=1}^{i} \sum_{j'=1}^{j} a_{i',j'} \\
&= \sum_{i'=1}^{i-1} \sum_{j'=1}^{j} a_{i',j'} + \sum_{i'=1}^{i} \sum_{j'=1}^{j-1} a_{i',j'} - \sum_{i'=1}^{i-1} \sum_{j'=1}^{j-1} a_{i',j'} + a_{i,j} \\
&= S_{i-1,j} + S_{i,j-1} - S_{i-1,j-1} + a_{i,j}
\end{aligned}
$$

可以认为，边界条件为当 $i = 0$ 或 $j = 0$ 时， $S_{i,j} = 0$；

递推即可求。

### 示例代码

```c
#include <stdio.h>

typedef long long ll;

int n,m;
ll a[2010][2010];
ll s[2010][2010];
int main(){
    scanf("%d%d",&n,&m);
    for(int i=1;i<=n;i++)
        for(int j=1;j<=m;j++)
            scanf("%lld",&a[i][j]);
    for(int i=1;i<=n;i++)
        for(int j=1;j<=m;j++)
            s[i][j]=s[i-1][j]+s[i][j-1]-s[i-1][j-1]+a[i][j];
    int q;
    scanf("%d",&q);
    for(int id=1;id<=q;id++){
        int i1,j1,i2,j2;
        scanf("%d%d%d%d",&i1,&j1,&i2,&j2);
        printf("%lld\n",s[i2][j2]-s[i1-1][j2]-s[i2][j1-1]+s[i1-1][j1-1]);
    }
    return 0;
}
```

## F 木棍木棍

| 难度 | 考点 |
| --- | --- |
| 4 | 二分查找 |

### 题目

#### 题目描述

给定一个已经从小到大排好序的数组和 $q$ 次询问，每次询问给出一个数 $x$ ，求数组中等于 $x$ 的元素个数。

#### 输入格式

第一行一个整数 $n$ 。接下来 $n$ 个整数，按从小到大排列，用空格或换行分隔均可。接下来一行一个整数 $q$ 。接下来 $q$ 行，每行一个整数 $x$ 。

#### 输出格式

$q$ 行，每行一个整数，表示数组中等于 $x$ 的元素个数。

#### 样例输入

```text
6
1 2 2 2 3 5
3
2
4
1
```

#### 样例输出

```text
3
0
1
```

#### 数据范围

$n, q \le 5 \times 10^{5}$ 。每次询问逐个遍历会超时，需要用二分查找求出第一个 $\ge x$ 的位置和第一个 $> x$ 的位置。

### 答案

### 题目分析

一种最暴力的想法是，遍历整个数组并计数，这样的时间复杂度是 $O(nq)$，显然是不能通过本题的。

结合二分查找容易想到一个优化，二分查找到数字 $x$ 出现的一个位置，然后从这个位置分别向左向右遍历并计数。可是在最坏情况下，整个数组仅有一个值，那么这种方法仍然会达到 $O(nq)$ 的时间复杂度，仍然不能通过本题。

本题的正解是：先做一次二分查找找到数字 $x$ 第一次出现的位置，再二分查找找到数字 $x$ 最后一次出现的位置，然后就无须遍历数组，直接就能得到答案了。

具体实现的时候，我们可以去找整个数组第一个 $\ge x$ 的数的位置（这是 $x$ 的 `lower_bound`）和第一个 $> x$ 的数的位置（这是 $x$ 的 `upper_bound`）。

验证自己的二分查找写的对不对的时候，注意测试边界情况：数组最小值、数组最大值、比数组最小值更小的值、比数组最大值更大的值。

### 示例代码

```c
#include <stdio.h>
int a[500005];
int main(void)
{
    int n;
    scanf("%d", &n);
    for (int i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }
    int q;
    scanf("%d", &q);
    while (q--)
    {
        int x;
        scanf("%d", &x);
        int l = -1, r = n;
        while (r - l > 1)
        {
            int mid = (l + r) / 2;
            if (a[mid] < x)
                l = mid;
            else
                r = mid;
        }
        int lft = r; // a[lft] >= x
        l = -1, r = n;
        while (r - l > 1)
        {
            int mid = (l + r) / 2;
            if (a[mid] <= x)
                l = mid;
            else
                r = mid;
        }
        int rt = r; // a[rt] > x
        printf("%d\n", rt - lft);
    }
    return 0;
}
```

## G 我爱跳台阶

| 难度 | 考点 |
| --- | --- |
| 4 | 动态规划 |

### 题目

#### 题目描述

有 $n$ 级台阶，一开始站在底部，每次可以向上迈 1 到 $k$ 级台阶。求恰好到达第 $n$ 级台阶的不同方式数，答案对 100003 取模。

#### 输入格式

一行两个整数 $n$ 和 $k$ ，用空格分隔。

#### 输出格式

一行一个整数，表示方式数对 100003 取模的结果。

#### 样例输入

```text
5 2
```

#### 样例输出

```text
8
```

#### 数据范围

$n \le 10^{5}$ 。用双重循环枚举前 $k$ 项之和会超时，需要动态维护最近 $k$ 项的和。

### 答案

### 题目分析

这道题不同于一般的斐波那契数列，我们需要推导一下公式。

原版公式：

$$
\begin{aligned}
F_i &= F_{i-1} + F_{i-2}(i > 2) \\
F_i &= 1(i = 1 或 i = 2)
\end{aligned}
$$

这种斐波那契数列可以解决以下问题

有 $N$ 级的台阶，你一开始在底部，每次可以向上迈 $1$ 级， $2$ 级台阶，问到达第 $N$ 级台阶有多少种不同方式。

这种问题我们的第 $i$ 个台阶，只能从第 $i-1$ 个台阶和第 $i-2$ 个台阶上走来，所以公式为上。

反过来看这道题目：

有 $N$ 级的台阶，你一开始在底部，每次可以向上迈最多 $K$ 级台阶，问到达第 $N$ 级台阶有多少种不同方式。

我们的第 $i$ 个台阶，就可以从第 $i-1, i-2$ 到 $i-k$ 个台阶登陆（ $i \ge k$），这就好办了

$$
F_i = \sum_{t=1}^{\min(i,k)} F_{i-t}
$$

注意事项

注意到数据范围在 $10^5$，若使用双层循环会 TLE，所以需要动态维护最后 $k$ 个数的和。

### 示例代码

```c
#include <ctype.h>
#include <math.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main()
{
    long long n, k;
    scanf("%lld%lld", &n, &k);
    long long c[100005];
    long long mod = 100003;
    c[0] = 1;
    long long sum = 0; // sum维护了最后k个数的和
    for (long long i = 1; i <= n; i++)
    {
        sum += c[i - 1]; // 添加当前的数
        sum %= mod;
        if (i > k)
        {
            sum += mod;
            sum -= c[i - k - 1]; // 将开始的数减去
            sum %= mod;
        }
        c[i] = sum % mod;
    }
    printf("%lld", c[n] % mod);
    return 0;
}
```

## H 括号序列

| 难度 | 考点 |
| --- | --- |
| 5 | DFS |

### 题目

#### 题目描述

求由 $n$ 对括号构成的所有合法括号序列（任意前缀中左括号数量都不少于右括号数量），按字典序从小到大输出前 $k$ 个。

#### 输入格式

一行两个整数 $n$ 和 $k$ ，用空格分隔。

#### 输出格式

每行一个长度为 $2n$ 的合法括号序列，按字典序输出，最多输出 $k$ 个；合法序列不足 $k$ 个时全部输出。

#### 样例输入

```text
2 2
```

#### 样例输出

```text
(())
()()
```

#### 数据范围

需要在搜索过程中剪枝（剩余位置数小于未匹配左括号数时直接返回），否则可能超时。

### 答案

### 题目分析

考虑使用深度优先搜索（DFS）来递归构造所有合法括号序列，利用递归的自然遍历顺序保证按字典序输出。

我们如下定义递归状态：

- `cur`：当前填到的位置（从 1 开始）；
- `left`：当前未匹配的左括号数；
- `a[cur]`：当前位置字符。

搜索过程中遵循以下规则：

1. 每次优先尝试放 `'('`；
2. 若已有未匹配的左括号（`left > 0`），则可放 `')'`；
3. 当填满 $2n$ 个字符且 `left == 0` 时，输出一个合法序列；
4. 若已输出 $k$ 个序列，则终止搜索。

有一个很有效的剪枝方法：每次搜索时判断剩余位置数是否小于未匹配左括号数，如果是，则可以提前返回。未采取有效剪枝手段的做法可能会超时。

### 示例代码

```c
#include <stdio.h>

#define N 200005

int n, k, cnt, flag;
char a[N];

void DFS(int cur, int left) {
    if (flag) return;
    if (2 * n - cur + 1 < left) return;
    if (cur == 2 * n + 1) {
        if (left == 0) {
            for (int i = 1; i <= 2 * n; i++) {
                putchar(a[i]);
            }
            putchar('\n');
            cnt++;
            if (cnt == k) {
                flag = 1;
            }
        }
        return;
    } else {
        a[cur] = '(';
        DFS(cur + 1, left + 1);
        a[cur] = ')';
        if (left > 0) {
            DFS(cur + 1, left - 1);
        }
    }
}

signed main() {
    scanf("%d%d", &n, &k);
    DFS(1, 0);
    return 0;
}
```

## I 愤怒的老鸟

| 难度 | 考点 |
| --- | --- |
| 6 | 动态规划，多维数组 |

### 题目

#### 题目描述

道路上有 $N$ 个格子，编号 1 到 $N$ ，第 $i$ 个格子上有 $coin_i$ 枚金币； $stone_i$ 为 1 表示该格是石头，为 0 表示是安全格。小鸟从第 1 格出发，先获得第 1 格的金币。它手上有 1 步、2 步、4 步、8 步的飞行卡各若干张，必须把这些卡全部用完：每用一张卡就向前移动相应的步数，落在石头上则当前金币数减半（向下取整），落在安全格则金币数增加该格的金币数。求最终能拥有的最大金币数。

#### 输入格式

第一行两个整数 $N$ 和 $M$ 。第二行 $N$ 个整数，表示各格的金币数。第三行四个整数，依次表示 1 步、2 步、4 步、8 步飞行卡的张数。第四行 $N$ 个整数，表示各格是否为石头（1 表示石头，0 表示安全格）。

#### 输出格式

一行一个整数，表示能获得的最大金币数。

#### 样例输入

```text
5 5
1 2 3 4 5
1 1 0 0
0 0 1 0 0
```

#### 样例输出

```text
7
```

#### 数据范围

每种飞行卡的张数不超过 40。

### 答案

### 题目分析

这题看似是路径问题，其实是一个“有限资源 + 顺序影响结果”的组合优化问题，非常适合用动态规划。

我们之所以要用 DP，是因为：

- 每次使用哪种飞行卡，会改变当前位置与当前金币；
- 同一张卡的顺序不同，结果也不同；
- 但若我们知道“每种卡用了几张”时的最大金币，就能在此基础上继续扩展，不必重复计算。

因此我们用四维 DP：`dp[a][b][c][d]` 表示用了 `a, b, c, d` 张 $1$、 $2$、 $4$、 $8$ 步卡后，能获得的最大金币数。

状态天然具有“无后效性”——未来决策只与当前卡使用情况和当前位置有关，和历史顺序无关，所以可以用 DP。

转移时，从当前状态尝试再使用一张飞行卡：

- 若落在石头上，金币减半；
- 若落在安全格，金币相加。

不断更新状态的最大值即可。

最终所有卡都用完的状态，就是答案。

这种 DP 的特点是：

- 状态有限、可枚举；
- 转移明确、重复子问题多；
- 复杂度 `O(41^4)` 可接受。

### 示例代码

```c
#include <stdio.h>
#include <string.h>

int coin[1005];
int stone[1005];
int dp[41][41][41][41];

int max(int a, int b) { return a > b ? a : b; }

int main() {
    int N, M;
    scanf("%d%d", &N, &M);
    for (int i = 1; i <= N; ++i) scanf("%d", &coin[i]);
    int cnt1, cnt2, cnt4, cnt8;
    scanf("%d%d%d%d", &cnt1, &cnt2, &cnt4, &cnt8);
    for (int i = 1; i <= N; ++i) scanf("%d", &stone[i]);

    for (int a = 0; a <= 40; ++a)
        for (int b = 0; b <= 40; ++b)
            for (int c = 0; c <= 40; ++c)
                for (int d = 0; d <= 40; ++d)
                    dp[a][b][c][d] = -1;

    dp[0][0][0][0] = coin[1]; // 起点金币

    for (int a = 0; a <= cnt1; ++a)
        for (int b = 0; b <= cnt2; ++b)
            for (int c = 0; c <= cnt4; ++c)
                for (int d = 0; d <= cnt8; ++d) {
                    int cur = dp[a][b][c][d];
                    if (cur == -1) continue;
                    int pos = 1 + a + 2*b + 4*c + 8*d;
                    if (pos > N) continue;

                    if (a < cnt1) {
                        int np = pos + 1;
                        if (np <= N) {
                            int next = stone[np] ? cur / 2 : cur + coin[np];
                            dp[a + 1][b][c][d] = max(dp[a + 1][b][c][d], next);
                        }
                    }
                    if (b < cnt2) {
                        int np = pos + 2;
                        if (np <= N) {
                            int next = stone[np] ? cur / 2 : cur + coin[np];
                            dp[a][b + 1][c][d] = max(dp[a][b + 1][c][d], next);
                        }
                    }
                    if (c < cnt4) {
                        int np = pos + 4;
                        if (np <= N) {
                            int next = stone[np] ? cur / 2 : cur + coin[np];
                            dp[a][b][c + 1][d] = max(dp[a][b][c + 1][d], next);
                        }
                    }
                    if (d < cnt8) {
                        int np = pos + 8;
                        if (np <= N) {
                            int next = stone[np] ? cur / 2 : cur + coin[np];
                            dp[a][b][c][d + 1] = max(dp[a][b][c][d + 1], next);
                        }
                    }
                }

    printf("%d\n", dp[cnt1][cnt2][cnt4][cnt8]);
    return 0;
}
```

## J 小球排列

| 难度 | 考点 |
| --- | --- |
| 7-8 | 动态规划 |

### 题目

#### 题目描述

有 $k$ 种小球，编号 0 到 $k - 1$ 。给定一个长度为 $n$ 的模板串，其中一部分位置已经写好了小球的编号（数字 0 到 $k - 1$ ），其余位置是下划线 `_` ，可以任意填入一种小球的编号。填好之后，每种出现过的小球的价值等于它最后一次出现的位置。要求所有出现过的小球的价值之和恰好等于 $m$ 。求满足条件的填法数，答案对 998244353 取模。

#### 输入格式

第一行两个整数 $m$ 和 $k$ ，用空格分隔。第二行一个字符串，表示模板，其长度即为 $n$ ，由数字 0 到 $k - 1$ 和下划线组成。

#### 输出格式

一行一个整数，表示满足条件的填法数对 998244353 取模的结果。

#### 样例输入

```text
4 2 0_1
```

#### 样例输出

```text
1
```

### 答案

### 题目分析

易得一种小球的总价值是其最后一次出现的位置，从而容易转化为类似背包问题的方法，我们可以利用状态压缩判断后续某个数字是否还会出现。如果状态表示这一位可以填的数字后续不会出现了，则我们可以选取这一位作为它最后一次出现，背包减去其位置即可。

顺序遍历，由于每一位可以出现的数字可以是自由的（下划线），也可以是强制的（数字），因此需要对每一位是数字还是下划线分类判断。

- 如果是数字：这一位填法是固定的。如果它是最后一次强制出现，我们可以自由选择它后续是否出现。
- 如果是下划线：后续还会出现的数字这里都可以填；同时也可以作为最后一次出现。

从而推出状态转移方程。由于情况较为复杂，故详见以下代码。

此方法的时间复杂度为 $O(nmk2^k)$，可以通过 $50\%$ 的测试点。

### 0.5 分代码

```c
#include <stdio.h>
#include <string.h>
const int mod = 998244353;

char str[100];
long long dp[81][128][540]; // dp[i][j][k] 表示：遍历到第 i 位时，状态压缩为 j（第 x 位表示数字 x 后面不会出现），背包剩余空间为 k 的状态数
int stop[10]; // 表示输入序列中，每个数字最后出现的位置

int popcnt(int x) { // 返回 x 二进制表示中 1 的个数
    int ret = 0;
    while ( x > 0 ) {
        ret += x & 1;
        x >>= 1;
    }
    return ret;
}

int main() {
    int n, m, k;
    scanf("%d%d%s", &m, &k, str + 1); // 从下标 1 开始符合本题背景
    n = strlen(str + 1);
    const int snum = 1 << k; // 状态数

    for ( int i = 1; i <= n; i++ )
        if ( str[i] != '_' ) {
            str[i] -= '0'; // '_' = 95，因此不会出现冲突
            stop[str[i]] = i; // 更新最后一次出现的位置
        }
    
    for ( int status = 0; status < snum; status++ ) {
        int tmp = status;
        for ( int j = 0; j < k; j++ )
            if ( stop[j] > 0 )
                tmp &= ~(1 << j);
        dp[0][tmp][m] = 1;
    }

    for ( int i = 1; i <= n; i++ )
        for ( int l = 0; l <= m; l++ ) {
            if ( str[i] != '_' ) // str[i] 是数字
                for ( int status = 0; status < snum; status++ ) {
                    dp[i][status][l] = (dp[i][status][l] + dp[i - 1][status][l]) % mod; // 不是最后一次出现
                    if ( ((status >> str[i]) & 1) == 0 && stop[str[i]] == i && l >= i ) // 之前状态认为会出现，当前为是强制出现的最后一次，则可以选取为最后一次出现
                        dp[i][status | (1 << str[i])][l - i] = (dp[i][status | (1 << str[i])][l - i] + dp[i - 1][status][l]) % mod;
                }
            else
                for ( int status = 0; status < snum; status++ ) {
                    dp[i][status][l] = (dp[i][status][l] + dp[i - 1][status][l] * (k - popcnt(status))) % mod; // 后续还可能出现的所有数都可以填
                    for ( int j = 0; j < k; j++ )
                        if ( ((status >> j) & 1) == 0 && stop[j] < i && l >= i ) // j 之前状态认为还会出现，后续没有强制出现，则可以选取为最后一次出现
                            dp[i][status | (1 << j)][l - i] = (dp[i][status | (1 << j)][l - i] + dp[i - 1][status][l]) % mod;
                }
        }

    printf("%lld", dp[n][snum - 1][0]); // 到第 n 位，后面所有数字都不会出现，背包剩余空间为 0 则是答案
    return 0;
}
```

### 满分代码

由于我们只关心后面还会出现几种数字，并不关系具体是什么数字，因此状态压缩是不必要的，我们只需要讨论可以填的数字种类即可。

同样顺序遍历，对每一位进行分类讨论，不过我们这次只关心下划线处能填几种数字，而不关心具体数字是什么。

初始化时，假设某状态认为总共会出现 $k''$ 种数字，强制出现 $k'$ 种数字，那么我们可以在剩下的 $k - k'$ 个数字中挑选 $k'' - k'$ 个数字填入序列中。由于我们的答案总是来源于初始条件的累加，因此要使使用 $k''$ 种数字的结果翻组合数 $C_{k-k'}^{k''-k'}$，我们只需要让初值从 $1$ 变成该组合数即可。

此方法的时间复杂度为 $O(nmk)$，可以通过全部测试点。

注：由于内存限制，我们需要使用滚动数组，详见代码。

```c
#include <stdio.h>
#include <string.h>
const int mod = 998244353;

char str[1010];
int cnt[1010], isstop[1010]; // cnt[i] 表示从第 i 位开始，强制出现的数字数；isstop 表示是否为某个数最后一次强制出现
long long dp[2][11][10000]; // dp[i % 2][j][k] 表示遍历到第 i 位，后面还会出现 j 种数字，背包剩余容量为 k 的种类数

int popcnt(int x) { // 返回 x 二进制表示中 1 的个数
    int ret = 0;
    while ( x > 0 ) {
        ret += x & 1;
        x >>= 1;
    }
    return ret;
}
long long C(int n, int m) { // 计算组合数
    long long ret = 1;
    for ( int i = 1; i <= m; i++ )
        ret = ret * (n - i + 1) / i;
    return ret;
}

int main() {
    int n, m, k;
    scanf("%d%d%s", &m, &k, str + 1);
    n = strlen(str + 1);

    for ( int i = n; i > 0; i-- ) 
        if ( str[i] != '_' ) {
            str[i] -= '0';
            cnt[i] = cnt[i + 1] | (1 << str[i]); // 先状压避免重复计数
        }
        else
            cnt[i] = cnt[i + 1];
    for ( int i = n; i > 0; i-- ) {
        cnt[i] = popcnt(cnt[i]);
        if ( cnt[i] != cnt[i + 1] ) // 第 i 位开始和第 i + 1 位开始强制出现的数字种数不同，则第 i 位是该数字最后一次强制出现
            isstop[i] = 1;
    }

    for ( int j = cnt[1]; j <= k; j++ ) // cnt[1] 即是 k'
        dp[0][j][m] = C(k - cnt[1], j - cnt[1]); // 初始化

    for ( int i = 1, ii = 1, li = 0; i <= n; i++, li = ii, ii ^= 1 ) { // 滚动数组，ii = i & 1, li = ii ^ 1，分别表示当前位和上一位（均模二）
        for ( int j = 0; j <= k; j++ )
            for ( int l = 0; l <= m; l++ )
                dp[ii][j][l] = 0; // 清零
        for ( int j = 0; j <= k; j++ )
            if ( str[i] != '_' )
                for ( int l = 0; l <= m; l++ ) {
                    dp[ii][j][l] = (dp[ii][j][l] + dp[li][j][l]) % mod;
                    if ( j > cnt[i + 1] && isstop[i] && l >= i )
                        dp[ii][j - 1][l - i] = (dp[ii][j - 1][l - i] + dp[li][j][l]) % mod;
                }
            else
                for ( int l = 0; l <= m; l++ ) {
                    dp[ii][j][l] = (dp[ii][j][l] + dp[li][j][l] * j) % mod;
                    if ( j > cnt[i + 1] && l >= i )
                        dp[ii][j - 1][l - i] = (dp[ii][j - 1][l - i] + dp[li][j][l] * (j - cnt[i + 1])) % mod; // 后续强制出现的数字不可能在这一位最后一次出现
                }
    }

    printf("%lld", dp[n & 1][0][0]); // 后续不会出现任何数字，且背包剩余空间为 0 时，即为所求答案
    return 0;
}
```
